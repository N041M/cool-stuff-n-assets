Deep sky

Milky Way

Preparing

Read me

Use it to put the Milky Way, a nebula or the black hole at its centre on your page, drawn in text characters on a canvas, as a hero picture or a background. The core on its own gives the objects as clouds of points and works out the black hole's light, for drawing them your own way.

Add it to your page

  1. Copy deep-sky.js, deep-sky-glyphs.js and glyph-renderer.js from the glyph-renderer folder next to your page.

  2. Add a canvas with a dark background and load the scripts at the end of the body.

    <canvas id="sky" style="background:#050608"></canvas>
<script src="glyph-renderer.js"></script>
<script src="deep-sky.js"></script>
<script src="deep-sky-glyphs.js"></script>

  3. Make the galaxy's stars a few milliseconds a frame, then draw it face-on with 40 kiloparsecs from the top of the canvas to the bottom.

    <script>
  const g = GlyphRenderer.create(document.getElementById('sky'));
  g.resize(innerWidth, innerHeight, devicePixelRatio, 6);
  g.view.k = g.H / 40;
  g.view.span = 40;
  requestAnimationFrame(function make() {
    if (!DeepSky.galaxy.prepare(10)) return requestAnimationFrame(make);
    g.clear();
    DeepSky.drawGalaxy(g);
    g.compose();
    g.present();
  });
</script>

  4. For a nebula or the black hole, draw it in place of the galaxy, with its middle and size in device pixels on the canvas. A nebula's points are made first with DeepSky.nebula.prepare(key, ms), as the galaxy's stars are. The black hole needs nothing made first.

    // the Crab Nebula, a third of the canvas's height across
DeepSky.drawNebula(g, { key: 'crab', x: g.W / 2, y: g.H / 2, radius: g.H / 6,
  frame: { u: [1, 0, 0], v: [0, 1, 0], w: [0, 0, 1] } });
// the black hole, its disc tipped 78° from face-on
DeepSky.drawHole(g, { x: g.W / 2, y: g.H / 2, radius: g.H / 6,
  frame: { N: [0, 0.98, 0.21], Q: [1, 0, 0], E: [0, 0.21, -0.98] } });

Draw the objects your own way

Copy only deep-sky.js, which needs nothing else. DeepSky.galaxy.points() and DeepSky.nebula.points(key) give the points as arrays of positions and colours. This draws a third of the galaxy's stars face-on, 10 pixels to a kiloparsec.

<canvas id="disc" width="400" height="400" style="background:#000"></canvas>
<script src="deep-sky.js"></script>
<script>
  requestAnimationFrame(function make() {
    if (!DeepSky.galaxy.prepare(10)) return requestAnimationFrame(make);
    const P = DeepSky.galaxy.points(), c = document.getElementById('disc').getContext('2d');
    for (let i = 0; i < P.n; i += 3) {
      c.fillStyle = `rgb(${P.col[i * 3]}, ${P.col[i * 3 + 1]}, ${P.col[i * 3 + 2]})`;
      c.fillRect(200 + P.pos[i * 3] * 10, 200 - P.pos[i * 3 + 1] * 10, 1, 1);
    }
  });
</script>

DeepSky.hole.at gives the black hole's light at any point round it, with the paths of light bent round the hole. This draws it one pixel at a time, 80 pixels to the radius of its ring.

<canvas id="hole" width="400" height="400"></canvas>
<script src="deep-sky.js"></script>
<script>
  const H = DeepSky.hole, o = {}, c = document.getElementById('hole').getContext('2d');
  const img = c.createImageData(400, 400);
  const s = H.setup({ N: [0, 0.98, 0.21], Q: [1, 0, 0], E: [0, 0.21, -0.98] },
                    { R: [1, 0, 0], U: [0, 1, 0], B: [0, 0, 1] }, 1 / 80);
  for (let y = 0, i = 0; y < 400; y++) for (let x = 0; x < 400; x++, i += 4) {
    H.at(s, (x - 200) / 80, (200 - y) / 80, o);
    img.data[i] = 255 * (o.r + o.f * H.colour[0]);
    img.data[i + 1] = 255 * (o.g + o.f * H.colour[1]);
    img.data[i + 2] = 255 * (o.b + o.f * H.colour[2]);
    img.data[i + 3] = 255;
  }
  c.putImageData(img, 0, 0);
</script>
