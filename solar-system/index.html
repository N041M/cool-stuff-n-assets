Planets where they are today

Use it to place the planets on a date in your own page, for example in a map of the Solar System or a sky chart. It gives positions as numbers, and your own code draws them.

What you need

  • planet-positions/ephemeris.js, 41 KB. It has no dependencies.

The code

planet-positions/ephemeris.js

// orbital elements at T (Julian centuries from J2000)
function elementsAt(name, T) {
  var e0 = ELEMENTS[name][0], r = ELEMENTS[name][1];
  var a = e0[0] + r[0] * T, e = e0[1] + r[1] * T, I = (e0[2] + r[2] * T) * DEG;
  var L = e0[3] + r[3] * T, peri = e0[4] + r[4] * T, node = e0[5] + r[5] * T;
  return { a: a, e: e, I: I, L: L * DEG, w: (peri - node) * DEG, node: node * DEG, M: norm360(L - peri) * DEG };
}

function solveKepler(M, e) {
  if (M > Math.PI) M -= 2 * Math.PI;
  var E = M + e * Math.sin(M);
  for (var i = 0; i < 8; i++) E -= (E - e * Math.sin(E) - M) / (1 - e * Math.cos(E));
  return E;
}

// position on an orbit at eccentric anomaly E, in the ecliptic frame
function orbitPoint(el, E) {
  var xp = el.a * (Math.cos(E) - el.e), yp = el.a * Math.sqrt(1 - el.e * el.e) * Math.sin(E);
  var cw = Math.cos(el.w), sw = Math.sin(el.w), cn = Math.cos(el.node), sn = Math.sin(el.node);
  var ci = Math.cos(el.I), si = Math.sin(el.I);
  return [
    (cw * cn - sw * sn * ci) * xp + (-sw * cn - cw * sn * ci) * yp,
    (cw * sn + sw * cn * ci) * xp + (-sw * sn + cw * cn * ci) * yp,
    (sw * si) * xp + (cw * si) * yp
  ];
}

// a planet's position at jde (a Julian day in Terrestrial Time); for
// 'earth' the table gives the Earth-Moon barycentre
function planetAt(name, jde) {
  var el = elementsAt(name, (jde - 2451545.0) / 36525);
  return orbitPoint(el, solveKepler(el.M, el.e));
}
function planetPosition(name, date) { return planetAt(name, ttDay(date)); }

Each planet's orbit is six elements that change slowly over the centuries, from JPL's approximate elements. planetAt takes them at the date, solves Kepler's equation for the eccentric anomaly with Newton's method and turns the point on the ellipse into ecliptic coordinates in AU.

The same file places the Moon, 20 other moons, Pluto, Halley's Comet, seven spacecraft and the nearest stars, and gives how each body is turned.

Add it to your page

  1. Copy ephemeris.js next to your page.

  2. Load it before your own script.

    <script src="ephemeris.js"></script>

  3. Ask for a planet's position on a date. This finds how far Mars was from the Earth when it came closest in January 2025.

    const date = new Date('2025-01-12T00:00:00Z');
const mars = Ephemeris.planetPosition('mars', date);
const earth = Ephemeris.planetPosition('earth', date);
const au = Math.hypot(mars[0] - earth[0], mars[1] - earth[1], mars[2] - earth[2]);
console.log(au.toFixed(3) + ' AU');   // 0.642 AU

Planet maps

Use it to give a planet, moon or star in your own scene its surface. Each map is made in the browser from code, with no image files, and comes as RGB bytes for a canvas or a texture.

What you need

  • planet-surfaces/surfaces.js, 69 KB.
  • planet-surfaces/earth-map.js, 25 KB, for the Earth's map only. It is the Earth's land and ice from Natural Earth.

The code

planet-surfaces/surfaces.js

function fillRow(gen, L, y) {
  var lat = 90 - (y + 0.5) / L.h * 180, cl = Math.cos(lat * DEG), sl = Math.sin(lat * DEG);
  for (var x = 0; x < L.w; x++) {
    var lon = (x + 0.5) / L.w * 360 - 180;
    gen(lon, lat, cl * Math.cos(lon * DEG), cl * Math.sin(lon * DEG), sl, px3);
    var o = (y * L.w + x) * 3;
    L.d[o] = px3[0] * 255; L.d[o + 1] = px3[1] * 255; L.d[o + 2] = px3[2] * 255;
  }
}

// ...

// compute full-size rows for up to budgetMs; returns progress 0..1
function refine(key, budgetMs) {
  var T = texture(key);
  if (T.ready) return 1;
  var gen = spec(key).gen, top = T.levels[0], t0 = performance.now(), start = T.row;
  while (T.row < T.h && performance.now() - t0 < budgetMs) fillRow(gen, top, T.row++);
  chain(T.levels, start & ~7, (T.row + 7) & ~7);
  if (T.row >= T.h) T.ready = true;
  T.progress = T.row / T.h;
  return T.progress;
}

Each body has a function that gives the colour at a longitude and latitude, from noise, craters, bands and spots placed at their real coordinates. fillRow calls it for every pixel of one row of the map. refine fills the full-size map a few rows at a time within a budget of milliseconds, so a page stays responsive while a large map is made.

Add it to your page

  1. Copy surfaces.js next to your page. For the Earth, copy earth-map.js as well.

  2. Load them in the head, the Earth's map first.

    <script src="earth-map.js"></script>
<script src="surfaces.js"></script>

  3. Get a body's map at full size. The pixels are RGB bytes in d, row by row from the north pole, with east longitude running from −180° to 180° across.

    const map = Surfaces.texture("mars");
Surfaces.refine("mars", Infinity);
const { w, h, d } = map.levels[0];

  4. Copy the pixels into a canvas.

    const canvas = document.createElement("canvas");
canvas.width = w;
canvas.height = h;
const image = new ImageData(w, h);
for (let i = 0, j = 0; i < d.length; i += 3, j += 4) {
  image.data[j] = d[i];
  image.data[j + 1] = d[i + 1];
  image.data[j + 2] = d[i + 2];
  image.data[j + 3] = 255;
}
canvas.getContext("2d").putImageData(image, 0, 0);

The README also shows the map as a texture in three.js and in WebGL.

Drawing in characters

Use it to show a 3D shape, a logo or any other picture in text characters. Your code gives the light at each point of a canvas, and the renderer picks a character and a colour for every cell.

Every demo on this page has a Glyphs button. With the glyphs off, the renderer draws the same light as a smooth picture, one pixel a sample.

The renderer draws with WebGL where the browser has it and on a 2D canvas where it does not. The explorer below switches between the two while it runs, and the picture is the same on both.

What you need

  • glyph-renderer/glyph-renderer.js, 47 KB. It has no dependencies.
  • A canvas. WebGL is optional.

The code

glyph-renderer/glyph-renderer.js

var state = 0;
if (maxL >= 0.035) {
  var gi, mean = sumL / 6, spread = maxL - minL;
  if (spread > tune.edge && spread > tune.edgeRel * maxL) {
    var key = 0;
    for (k = 0; k < 6; k++) {
      var vr = v6[k] / maxL, q = maxL * (tune.contrast === 2 ? vr * vr : Math.pow(vr, tune.contrast));
      key |= Math.round(clamp(q, 0, 1) * Q7) << (k * 3);
    }
    gi = matchGlyph(key);
  } else {
    gi = rampLUT[clamp(Math.round(mean * 255), 0, 255)];
  }
  if (gi) {
    var aq = alphaLevel(mean);
    if (aq) state = gi | (rowFor(ar, ag, ab) << 7) | (aq << 15);
  }
}
if (!state && cOcc[cell] < 0.35 && bg) state = bg.state(cell);
putCell(cell, state);

Each cell is sampled at six points, two across and three down. Where the samples differ enough, the cell holds an edge and takes the glyph whose shape matches them best. Elsewhere it takes a glyph from a density ramp by its mean brightness, and its colour comes from a palette of 36 hues and 7 saturations.

glyph-renderer/glyph-renderer.js

mctx = document.createElement('canvas').getContext('2d');

The renderer asks the canvas for WebGL first. A canvas that already has a 2D context cannot give a WebGL one, so glStart fails and the renderer draws on the 2D canvas.

Add it to your page

  1. Copy glyph-renderer.js next to your page.

  2. Add a canvas and load the script at the end of the body. The characters are light, so give the canvas a dark background.

    <canvas id="art" style="background:#050608"></canvas>
<script src="glyph-renderer.js"></script>

  3. Make a renderer, size it and draw. The function you give draw gets a point in CSS pixels from the canvas's top left and writes the red, green and blue light there into out. This one draws a sphere lit from the upper left.

    <script>
  const g = GlyphRenderer.create(document.getElementById('art'));
  g.resize(320, 320, devicePixelRatio, 10);
  g.draw(function (x, y, out) {
    const dx = (x - 160) / 140, dy = (y - 160) / 140, z = 1 - dx * dx - dy * dy;
    if (z > 0) out[0] = out[1] = out[2] = Math.max(0, 0.6 * Math.sqrt(z) - 0.5 * dx - 0.5 * dy);
  });
</script>

  4. To draw on a 2D canvas instead of with WebGL, ask the canvas for its 2D context just before the renderer is made in step 3. g.gl is then false.

    document.getElementById('art').getContext('2d');
const g = GlyphRenderer.create(document.getElementById('art'));

  5. To show the same light as a smooth picture without characters, turn the glyphs off and call draw again with the same function. g.setGlyphs(true) turns them back on.

    g.setGlyphs(false);

The starry background

Use it as a starry background behind your page. The stars twinkle, the sky drifts slowly past, and your content stays on top of it.

What you need

  • night-sky/night-sky.js, 17 KB, and night-sky/night-sky-glyphs.js, 10 KB.
  • glyph-renderer/glyph-renderer.js, 47 KB, for the look in characters. night-sky.js on its own gives the stars as data.

The code

night-sky/night-sky-glyphs.js

function placeStars(list, base, offDisc, disc) {
  var W = T.W, H = T.H, cw = T.cw, ch = T.ch, cols = T.cols, view = T.view;
  var f = 0.95 * H, R = view.R, U = view.U, B = view.B, mask = offDisc && !!disc && Math.abs(dot(B, GAL_Z)) > 0.05;
  for (var i = 0; i < list.length; i++) {
    var d = list[i].d, zc = -(d[0] * B[0] + d[1] * B[1] + d[2] * B[2]);
    if (zc < 0.1) continue;
    var x = W / 2 + f * (d[0] * R[0] + d[1] * R[1] + d[2] * R[2]) / zc;
    var y = H / 2 - f * (d[0] * U[0] + d[1] * U[1] + d[2] * U[2]) / zc;
    if (x < 0 || y < 0 || x >= W || y >= H) continue;
    if (mask) { var gp = disc(x, y); if (smoothstep(4, 19, Math.sqrt(gp[0] * gp[0] + gp[1] * gp[1])) < list[i].cut) continue; }
    var cell = ((y / ch) | 0) * cols + ((x / cw) | 0);
    var cur = starAt[cell];
    if (cur && starOf(cur).a >= list[i].a) continue;
    if (!cur) starCells.push(cell);
    starAt[cell] = base + i + 1;
  }
}

// ...

// the packed state of a cell, for the renderer: 0 where there is no
// star, else its glyph, colour and opacity as it twinkles at L.time
// seconds, or held at its brightest while L.still is set
L.state = function (cell) {
  var si = starAt ? starAt[cell] : 0;
  if (!si) return 0;
  var sky = si <= STARS.length, j = sky ? si - 1 : si - 1 - STARS.length, st = sky ? STARS[j] : GSTARS[j], fade = sky ? starFade : gStarFade;
  var a = st.a * fade * (1 - st.amp + (L.still ? st.amp : st.amp * Math.sin(L.time * st.tw + st.ph)));
  var aq = Math.round(clamp(a, 0, 1) * 63);
  return aq ? T.pack(sky ? gStar[j] : gGal[j], sky ? rStar[j] : rGal[j], aq) : 0;
};

Each star is a direction in the sky. placeStars projects it through the renderer's camera into a character cell and keeps the brightest star in each cell. state gives that cell its glyph, colour and opacity as the star twinkles, and the renderer draws it in every cell that nothing else covers.

Add it to your page

  1. Copy night-sky.js, night-sky-glyphs.js and glyph-renderer.js from the glyph-renderer folder next to your page.

  2. Load them at the end of the head.

    <script defer src="glyph-renderer.js"></script>
<script defer src="night-sky.js"></script>
<script defer src="night-sky-glyphs.js"></script>

  3. Put the canvas first in the body.

    <canvas id="sky" aria-hidden="true"></canvas>

  4. Fix the canvas behind the page.

    body { margin: 0; background: #050608; }

main { position: relative; z-index: 1; }

  5. Start the sky at the end of the body.

    <script>
  addEventListener('DOMContentLoaded', () => {
    const g = GlyphRenderer.create(document.getElementById('sky'));
    const sky = NightSky.layer(g);
    const still = matchMedia('(prefers-reduced-motion: reduce)').matches;
    let lon = 4.7, laidAt = 0;
    // look along the ecliptic at longitude lon and draw the sky
    function draw() {
      const c = Math.cos(lon), s = Math.sin(lon);
      g.view.B = [-c, -s, 0]; g.view.R = [s, -c, 0]; g.view.U = [0, 0, 1];
      sky.place();
      g.clear();
      NightSky.drawGalaxies(g, 1);
      g.compose(sky);
      g.present();
    }
    function size() { g.resize(innerWidth, innerHeight, devicePixelRatio, 10); draw(); }
    addEventListener('resize', size);
    sky.still = still;
    size();
    if (still) return;
    requestAnimationFrame(function frame(ms) {
      sky.time = ms / 1000;
      // one turn in 15 minutes, laid out again 4 times a second
      if (ms - laidAt > 250) { laidAt = ms; lon = 4.7 + ms / 1000 * 0.007; draw(); }
      else { g.update(sky, true); g.present(); }
      requestAnimationFrame(frame);
    });
  });
</script>

Nebulae, the galaxy and the black hole

Use it to put the Milky Way, a nebula or the black hole at its centre on your page, as a large picture at the top or as a background.

What you need

  • deep-sky/deep-sky.js, 58 KB, and deep-sky/deep-sky-glyphs.js, 16 KB.
  • glyph-renderer/glyph-renderer.js, 47 KB, for the look in characters. deep-sky.js on its own gives the objects as points and the black hole's light.

The code

deep-sky/deep-sky-glyphs.js

// each star goes into the one sample it falls in, so it stays a crisp dot
var cols2 = cols * 2, rows3 = rows * 3, isx = 2 / cw, isy = 3 / ch;
for (i = 0; i < Pt.n; i += stride) {
  var o3 = i * 3, gx = pos[o3], gy = pos[o3 + 1], gz = pos[o3 + 2];
  var sx = ((ox + xx * gx + xy * gy + xz * gz) * isx) | 0;
  if (sx < 0 || sx >= cols2) continue;
  var sy = ((oy + yx * gx + yy * gy + yz * gz) * isy) | 0;
  if (sy < 0 || sy >= rows3) continue;
  var si = (((sy / 3) | 0) * cols + (sx >> 1)) * 6 + (sy % 3) * 2 + (sx & 1);
  bR[si] += pc[o3] * pg; bG[si] += pc[o3 + 1] * pg; bB[si] += pc[o3 + 2] * pg;
}

The galaxy is 450,000 stars, made from fixed seeds a few milliseconds at a time. Each star is projected through the renderer's camera, and its light goes into the one sample it falls in, so it stays a crisp dot. The nebulae are clouds of points drawn the same way, and the black hole's light is traced along paths bent round it.

Comet tails

Use it to show Halley's Comet, or another comet on an orbit you give it, on a page about the sky. Its coma and tails grow and turn with the date.

What you need

  • comet/comet.js, 16 KB, and comet/comet-glyphs.js, 8 KB.
  • glyph-renderer/glyph-renderer.js, 47 KB, for the look in characters. comet.js on its own gives the tails as points.

The code

comet/comet.js

function dustNow(key, el, jd) {
  if (dust.key === key && Math.abs(jd - dust.jd) < 1 / 48) return dust;
  var nb = DUST_BETA.length, now = smallPosition(el, jd);
  var off = new Float64Array(DUST_N * nb * 3), w = new Float64Array(DUST_N);
  for (var i = 0; i < DUST_N; i++) {
    var age = dustAge(i), st = cometState(el, jd - age), ia2 = 1 / (age * age);
    w[i] = tailActivity(Math.sqrt(dot(st[0], st[0])));
    for (var j = 0; j < nb; j++) {
      var q = st[0].slice(), v = st[1].slice(), n = Math.max(4, Math.ceil(age / 0.75)), o = (i * nb + j) * 3;
      for (var k = 0; k < n; k++) orbitStep(q, v, age / n, GM_SUN * (1 - DUST_BETA[j]));
      off[o] = (q[0] - now[0]) * ia2; off[o + 1] = (q[1] - now[1]) * ia2; off[o + 2] = (q[2] - now[2]) * ia2;
    }
  }
  dust = { key: key, jd: jd, off: off, w: w };
  return dust;
}

Dust grains leave the nucleus over the last 45 days. Sunlight pushes each grain outward, so it moves on an orbit of its own under a share of the Sun's pull. dustNow follows those orbits from each grain's start to the date, and the tail fans out and curves back along the comet's path. The ion tail points straight away from the Sun, tipped by the comet's speed.

  3. Add a canvas, load the three scripts and draw the comet. This puts the nucleus near the top left of the window, with a tenth of an AU to the window's height.

Spacecraft

Use it to show a spacecraft at its real size on a page about space missions. Voyager, New Horizons, Pioneer, Parker Solar Probe and JWST are small 3D models built in metres.

What you need

  • spacecraft/spacecraft.js, 23 KB, and spacecraft/spacecraft-glyphs.js, 6 KB.
  • glyph-renderer/glyph-renderer.js, 47 KB, for the look in characters. spacecraft.js on its own gives the models and a function that casts a ray through them.

The code

spacecraft/spacecraft-glyphs.js

var rayParts = SC.rayParts;
for (var r = r0; r < r1; r++) {
  for (var c = c0; c < c1; c++) {
    var cell = r * cols + c, occ = cOcc[cell], gc = (r - r0) * gw + c - c0, i0 = cellAt[gc], i1 = cellAt[gc + 1];
    if (i0 === i1) continue;
    for (var k = 0; k < 6; k++) {
      var i = cell * 6 + k, x0 = sampleX(c, k) - sx0, y0 = sy0 - sampleY(r, k), cov = 0, cr = 0, cg = 0, cb2 = 0;
      for (var j = 0; j < n; j++) {
        var x = fine ? x0 + ((j & 1) ? ax : -ax) : x0, y = fine ? y0 + ((j & 2) ? ay : -ay) : y0;
        cov += rayParts(parts, cellIds, i0, i1, x * R[0] + y * U[0], x * R[1] + y * U[1], x * R[2] + y * U[2], d, L, px, so);
        cr += so[0]; cg += so[1]; cb2 += so[2];
      }
      cov *= A / n;
      if (cov < 0.002) continue;
      if (dK[i]) T.freeze(i);
      var f = A / n;
      bR[i] = bR[i] * (1 - cov) + cr * f; bG[i] = bG[i] * (1 - cov) + cg * f; bB[i] = bB[i] * (1 - cov) + cb2 * f;
      if (cov > occ) occ = cov;
    }
    cOcc[cell] = occ;
  }
}

Each sample casts four rays through the model, or one while the camera moves, and tries only the parts whose outlines reach its cell. A ray keeps every part it passes, so a boom thinner than a sample covers some of the rays and shows as a faint line.

Flying the camera

Use the whole explorer as a page of its own, with buttons or links on your page that fly the camera from one body to another.

What you need

  • js/orrery.js, 139 KB, the explorer's engine.
  • The scripts of the seven parts above: planet-positions/ephemeris.js, planet-surfaces/earth-map.js and surfaces.js, glyph-renderer/glyph-renderer.js, night-sky/night-sky.js and night-sky-glyphs.js, deep-sky/deep-sky.js and deep-sky-glyphs.js, comet/comet.js and comet-glyphs.js, and spacecraft/spacecraft.js and spacecraft-glyphs.js. With orrery.js they come to about 470 KB.
  • WebGL where the browser has it. Without it the explorer draws on a 2D canvas.

The code

js/orrery.js

var RHO = 1.42;

// ...

var r2 = RHO * RHO, r4 = r2 * r2;
var b0 = (w1 * w1 - w0 * w0 + r4 * u1 * u1) / (2 * w0 * r2 * u1);
var b1 = (w1 * w1 - w0 * w0 - r4 * u1 * u1) / (2 * w1 * r2 * u1);
fl.r0 = -Math.asinh(b0);
fl.S = (-Math.asinh(b1) - fl.r0) / RHO;

// ...

// Where a flight is at progress p: its span w, the share of the way its
// target has come (f) and the share it has left (rest). f and rest are
// each worked out on their own, so neither loses precision near its end.
function flightAt(fl, p, out) {
  var c = Math.cos(Math.PI * p), s = fl.S * (0.5 - 0.5 * c), left = fl.S * (0.5 + 0.5 * c);
  if (fl.zoomOnly) { out.w = fl.w0 * Math.exp(fl.k * RHO * s); out.f = 1; out.rest = 0; return out; }
  var chx = Math.cosh(RHO * s + fl.r0), d = Math.sinh(RHO * fl.S) * chx;
  out.w = fl.w0 * Math.cosh(fl.r0) / chx;
  out.f = Math.sinh(RHO * s) * Math.cosh(fl.r0 + RHO * fl.S) / d;
  out.rest = Math.sinh(RHO * left) * Math.cosh(fl.r0) / d;
  return out;
}

A flight pans and zooms at once along the path van Wijk and Nuij worked out in 2003. It zooms out just enough to keep both ends in sight and moves at an even speed on the screen. S is the length of that path, and flightAt gives the size of the view and how far it has come at any point of the flight. The heading, pitch and roll ease to their new values over the same time.

Orrery.zoom and Orrery.rotate zoom and turn the view from your own controls. The explorer's demo.js ties them to the wheel, drags, pinches and keys.

Rings, shadows and eclipses

Use it to show a planet with its rings, the shadows the rings and the planet cast on each other, and eclipses at the moment they happen. This part lives in the explorer's engine.

The view above opens on Saturn in June 2017, when its rings were tilted furthest toward the Sun. The view below runs the total eclipse of the Moon of 7 September 2025 at a minute a second, over and over.

What you need

  • The explorer, set up as in Flying the camera. The rings come from planet-surfaces/surfaces.js, and the places of the Sun, the Moon and the planets from planet-positions/ephemeris.js.

The code

js/orrery.js

// shadow of the rings on the clouds, on the day side
if (Rg && Lz !== 0 && lit > 0) {
  var sx2 = -hzz * (1 - b.f) / Lz;
  if (sx2 > 0) {
    var qx = hx + sx2 * Lx, qy = hy + sx2 * Ly, rq = Math.sqrt(qx * qx + qy * qy);
    if (rq > Rg.inner && rq < Rg.outer) lit *= 1 - ringOp(Rg, rq, sampleR * 2) * 0.88;
  }
}

// ...

for (var si = 0; si < shd.length; si++) {
  var sd = shd[si], wx = hwx - sd.C[0], wy = hwy - sd.C[1], wz = hwz - sd.C[2];
  var sa = wx * sd.A[0] + wy * sd.A[1] + wz * sd.A[2];
  if (sa <= 0) continue;
  var pp = Math.sqrt(Math.max(0, wx * wx + wy * wy + wz * wz - sa * sa));
  var ru = sd.R - sa * sd.ku, rpn = sd.R + sa * sd.kp;
  ecl *= smoothstep(Math.max(ru, 0) - sd.R * 0.02, rpn, pp);
  // sunlight bent through the Earth's air lights the Moon red
  // inside the umbra, dimmest at the middle and brighter
  // toward the edge. It is far brighter here than in the sky,
  // so that a totally eclipsed Moon still gets characters.
  if (b.key === 'moon' && ru > 0) {
    var uu = pp / ru;
    red = Math.max(red, uu < 1 ? 0.22 + 0.25 * uu * uu : 0.47 * (1 - smoothstep(1, 1.5, uu)));
  }
}

// ...

// the planet's shadow across the rings
var b3 = px * Lx + py * Ly, sh = 1;
if (b3 < 0) sh = smoothstep(0.965, 1.02, Math.sqrt(Math.max(0, rr2 - b3 * b3 * ia3)));
var lt = ringLight * (0.10 + 0.90 * sh) * alpha;

Each sample on a planet looks back toward the Sun. Where that line crosses the ring plane inside the rings, the ring's opacity there dims the sample. A sample on the rings is dimmed where the line to the Sun passes through the planet.

A moon's shadow, or the Earth's on the Moon, is a cone of umbra inside a wider cone of penumbra, set by the size of the Sun and of the body that casts it. A sample's light falls from the outer edge of the penumbra to the edge of the umbra. Inside the Earth's umbra the Moon turns red, as sunlight bent through the Earth's air lights it.

  1. Set up the explorer as in Flying the camera, steps 1 to 3.

  2. Start it on Saturn on a date when its rings are open, or on the Moon during an eclipse. time is in milliseconds since 1970, and setWarp sets how many seconds of the clock pass each second.
