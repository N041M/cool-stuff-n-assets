STL models

Use it to let visitors turn, zoom and download 3D-printable models on a shop or project page. The models are STL files, so you can also print them or load one into a three.js scene of your own.

Features Models Looks The STL files

Models

Use the viewer to show a set of STL files on your page one at a time, with each model's size and a link to its file.

What you need

  • demo.js, looks.js and demo.css, with the fonts/ folder.
  • three.js r186 in vendor/three/, which holds three.module.js, three.core.js, STLLoader.js and OrbitControls.js.
  • Your STL files, or the ones in models/.
  • A browser with WebGL.
  • A web server. The models do not load when the page is opened straight from the file.

The code

demo.js

// Each model is fetched once. Its size is measured in the file's own axes,
// before it is turned to Y-up.
const loader = new STLLoader();
const cache = new Map();
function load(model) {
  let pending = cache.get(model.slug);
  if (!pending) {
    pending = fetch(model.file).then(function (res) {
      if (!res.ok) throw new Error(res.status + " " + res.statusText);
      return res.arrayBuffer();
    }).then(function (buffer) {
      const geometry = loader.parse(buffer);
      geometry.computeBoundingBox();
      const size = geometry.boundingBox.getSize(new THREE.Vector3());
      geometry.rotateX(-Math.PI / 2);
      geometry.computeBoundingBox();
      geometry.computeBoundingSphere();
      return {
        geometry: geometry,
        size: size,
        triangles: geometry.getAttribute("position").count / 3,
        bytes: buffer.byteLength,
      };
    });
    pending.catch(function () { cache.delete(model.slug); });
    cache.set(model.slug, pending);
  }
  return pending;
}

load() fetches each file once and parses it with STLLoader. It measures the model in the file's own axes and then gives it a quarter turn about x, because the files are Z-up and three.js is Y-up.

Reset sends the camera back to the view that show() saved for the model. The + and − buttons zoom with OrbitControls the way the mouse wheel does.

Add it to your page

  1. Copy demo.js, looks.js, demo.css and the folders fonts/, vendor/ and models/ next to your page.

  2. Add the import map, the stylesheet and the script to the head. The import map points three at the copy in vendor/three/.

    <script type="importmap">
  {
    "imports": {
      "three": "./vendor/three/three.module.js",
      "three/addons/": "./vendor/three/addons/"
    }
  }
</script>
<link rel="stylesheet" href="demo.css">
<script type="module" src="demo.js"></script>

  3. Copy the <main> element of demo.html into the body. Each item in its list is one model, so edit the list to show your own files. demo.js takes the model's id from data-model, its name from .pick-name and its file from the link.

    <li data-model="servo-skull-drone">
  <button type="button" class="pick" aria-pressed="false"><span class="idx label">09</span><span class="pick-name">Servo-Skull Drone</span><span class="pick-cat label">Characters</span></button>
  <a class="dl" href="models/servo-skull-drone.stl" download aria-label="STL file, servo-skull-drone.stl">STL</a>
</li>

Looks

Use the looks to show a model in the amber of an old tube screen, on a page with a retro or sci-fi style. They work on any model in a three.js scene.

What you need

  • looks.js, and three.js r186 from vendor/three/.
  • The --ph-* colours from demo.css.
  • A browser with WebGL.

The code

looks.js

void main() {
  // Nothing above the build height is drawn yet.
  if (vH > uReveal) discard;

  vec3 n = normalize(vNormalV);
  if (!gl_FrontFacing) n = -n;
  vec3 v = normalize(-vViewPos);

  // A key light, a soft fill and some light from above, all fixed to the
  // camera. The shade is pulled part of the way to six flat bands, and then
  // squared so the bands stay apart once the colour is brightened for the
  // screen.
  vec3 keyDir = normalize(vec3(0.45, 0.75, 0.55));
  vec3 fillDir = normalize(vec3(-0.75, 0.15, 0.35));
  float key = max(dot(n, keyDir), 0.0);
  float fill = max(dot(n, fillDir), 0.0);
  float hemi = 0.5 + 0.5 * n.y;
  float shade = key * 0.72 + fill * 0.16 + hemi * 0.2;
  shade = mix(shade, floor(shade * 6.0 + 0.5) / 6.0, 0.5);
  shade *= shade;

  vec3 halfDir = normalize(keyDir + v);
  float spec = pow(max(dot(n, halfDir), 0.0), 42.0);
  float fres = pow(1.0 - clamp(dot(n, v), 0.0, 1.0), 3.6);
  vec3 col = mix(uDark, uLit, clamp(shade, 0.0, 1.1));
  col += uHot * spec * 0.5;
  col += uRim * fres * 0.4;

  // Scanlines. They crawl upward while uTime runs.
  float scan = 1.0 - uScan * (0.5 + 0.5 * sin(gl_FragCoord.y * 1.35 - uTime * 5.0));
  col *= scan;

  float alpha = 1.0;
  if (uXray > 0.5) {
    float body = (0.05 + fres * 1.1) * uAmount;
    col = uRim * scan * mix(1.0, body, uGlow);
    alpha = clamp(body, 0.0, 1.0);
  }

  // The build line is a bright band just under the build height.
  float edge = (1.0 - smoothstep(0.0, 0.03, uReveal - vH)) * (1.0 - step(0.9999, uReveal));
  col = mix(col, uBuild, edge);
  alpha = max(alpha, edge);

  gl_FragColor = vec4(col, alpha);
  #include <colorspace_fragment>
}

One shader draws all three looks. It leaves out everything above the build height, pulls the shading towards six flat bands and darkens every few rows of pixels for the scanlines. X-ray keeps only the rim light and draws the model see-through.

Add it to your page

The viewer under Models has the looks already. These steps add them to a three.js scene of your own, starting from the page under The STL files.

  1. Copy looks.js next to your page.

  2. Add the colours to your stylesheet. looks.js reads them from the page. Set --bg to the colour behind the model.

    :root {
  --bg: #050608;
  --ph-dark: #080401;
  --ph-lit: #ffa035;
  --ph-rim: #ffdcae;
  --ph-hot: #fff4e4;
  --ph-build: #fff4e4;
  --ph-line: #d27a22;
  --ph-wire: #ffdcae;
  --ph-xray: #ffdcae;
  --ph-dim: #8a541a;
}

  3. Import createLooks at the top of the module script.

    import { createLooks } from "./looks.js";

  4. Replace the two lines that make the grey material and add the mesh with these lines. setLook() takes "phosphor", "wire" or "xray", and build() grows the model from the bottom.

    renderer.localClippingEnabled = true;
const looks = createLooks(scene);
looks.setColors(function (name) {
  return getComputedStyle(document.documentElement).getPropertyValue(name).trim();
});
looks.setModel({ geometry });
looks.setLook("phosphor");
looks.build();

  5. Replace the animation loop with this one, which moves the build and the scanlines on.

    const timer = new THREE.Timer();
renderer.setAnimationLoop(function (time) {
  timer.update(time);
  controls.update();
  looks.update(timer.getDelta(), true);
  renderer.render(scene, camera);
});

The format

  • The files are binary STL in millimetres, with Z up.
  • Each model is centred on x and y and rests on z = 0. Its front faces −y.
  • Every triangle carries its face normal.
  • Each file holds the whole model in one piece. It is made of closed parts that overlap where they meet, and no part floats.
  • The models have no split parts, magnet sockets or supports.

Print it

  1. Download a file from the table.

  2. Open it in your slicer. It comes in at its real size and stands the right way up.

  3. Add supports in the slicer where your printer needs them.

Show it on your page

The page needs WebGL and a web server, as the viewer does.

  1. Copy vendor/three/ and one file from models/, and add the import map from step 2 under Models to the head.

  2. Add a canvas and this script to the body. The files are Z-up, so the script turns the model to the Y-up of three.js.

    <canvas id="model"></canvas>
<script type="module">
  import * as THREE from "three";
  import { STLLoader } from "three/addons/loaders/STLLoader.js";
  import { OrbitControls } from "three/addons/controls/OrbitControls.js";

  const canvas = document.getElementById("model");
  const width = canvas.parentElement.clientWidth;
  const renderer = new THREE.WebGLRenderer({ canvas, antialias: true });
  renderer.setPixelRatio(Math.min(devicePixelRatio, 2));
  renderer.setSize(width, width * 3 / 4);

  const scene = new THREE.Scene();
  scene.add(new THREE.HemisphereLight(0xffffff, 0x555555, 2));
  const camera = new THREE.PerspectiveCamera(30, 4 / 3, 1, 5000);
  camera.add(new THREE.DirectionalLight(0xffffff, 1.5));
  scene.add(camera);
  const controls = new OrbitControls(camera, canvas);

  const geometry = await new STLLoader().loadAsync("models/servo-skull-drone.stl");
  geometry.rotateX(-Math.PI / 2);
  geometry.center();
  geometry.computeBoundingSphere();
  const material = new THREE.MeshStandardMaterial({ color: 0xb8b4ab, flatShading: true });
  scene.add(new THREE.Mesh(geometry, material));
  camera.position.set(0, 0, geometry.boundingSphere.radius * 4);

  renderer.setAnimationLoop(function () {
    controls.update();
    renderer.render(scene, camera);
  });
</script>
